{\small
\noindent
\begin{tabular}[t]{@{} l p{6.2cm} @{}}
\multicolumn{2}{@{}l}{\textbf{Symbols}}\\[2pt]
$\kappa$        & ISO~281 viscosity ratio, $\nuop/\nureq$ \\
$\nuop$         & kinematic viscosity at operating temperature [\si{mm^2\,s^{-1}}] \\
$\nureq$        & reference viscosity for full film formation [\si{mm^2\,s^{-1}}] \\
$n$             & rotational speed [rpm] \\
$d_{pw}$        & mean pitch diameter [mm] \\
$N$             & samples per segment \\
$x_n$           & voltage sample $n$ \\
$S_k$, $f_k$    & one-sided FFT magnitude and frequency at bin $k$ \\
$f_c$           & spectral centroid [Hz] \\
$\sigma_w$      & magnitude-weighted spectral bandwidth [Hz] \\
$\rho$          & Spearman correlation coefficient \\
\end{tabular}\hfill
